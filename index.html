function formatText() { let text = document.getElementById('raw-text').value; if(!text.trim()) { alert('변환할 내용을 입력해 주십시오.'); return; } // 1. 마크다운 표(Table)를 HTML 테이블로 변환하는 정교한 로직 let lines = text.split('\n'); let inTable = false; let tableHtml = ''; let processedLines = []; for (let i = 0; i < lines.length; i++) { let line = lines[i].trim(); // 표의 시작 또는 내부 행인 경우 (| 로 시작) if (line.startsWith('|') && line.endsWith('|')) { // 구분선 행(|---|---|)은 건너뜀 if (line.match(/^\|[:\-\|\s]+\|$/)) { continue; } if (!inTable) { inTable = true; tableHtml = ''; } let cols = line.split('|').slice(1, -1).map(c => c.trim()); // 첫 번째 행이 헤더인지 확인 (다음 줄이 구분선인지 체크) let isHeader = false; if (i + 1 < lines.length && lines[i+1].trim().match(/^\|[:\-\|\s]+\|$/)) { isHeader = true; } tableHtml += ''; cols.forEach(col => { let tag = isHeader ? 'th' : 'td'; let style = isHeader ? 'border:1px solid #ccc; padding:10px; background:#f2f4f6; font-weight:bold; text-align:center;' : 'border:1px solid #ccc; padding:10px; text-align:center;'; tableHtml += `<${tag} style="${style}">${col}`; }); tableHtml += ''; } else { if (inTable) { tableHtml += '
'; processedLines.push(tableHtml); tableHtml = ''; inTable = false; } processedLines.push(lines[i]); } } if (inTable) { tableHtml += ''; processedLines.push(tableHtml); } // 2. 전체 텍스트 조합 및 기타 마크다운 서식 변환 text = processedLines.join('\n'); // 불필요한 연속 기호 정리 text = text.replace(/\|\|\|+/g, '
'); // 인용구 변환 text = text.replace(/^>\s?(.*)$/gm, '
$1
'); // 링크 변환 text = text.replace(/\[(.*?)\]\((.*?)\)/g, '$1'); // 이미지 박스 변환 text = text.replace(/\[이미지\]/g, '
📸 네이버 블로그에서 이 위치에 이미지를 첨부하세요
'); // 굵은 글씨 및 기울임 text = text.replace(/\*\*(.*?)\*\*/g, '$1'); text = text.replace(/\*(.*?)\*/g, '$1'); // 3. 줄바꿈 처리 (HTML 태그가 아닌 일반 줄에만
적용) let finalLines = text.split('\n'); let formattedOutput = finalLines.map(line => { let trimmed = line.trim(); if (trimmed.startsWith(''; }); document.getElementById('output-area').innerHTML = formattedOutput.join('
'); }
